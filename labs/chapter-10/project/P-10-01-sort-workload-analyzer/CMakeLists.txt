cmake_minimum_required(VERSION 3.25)
project(sort_workload_lab LANGUAGES CXX)

if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LAB_USE_SOLUTION "Build the reviewed reference implementation" OFF)

if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_subdirectory(tasks/task-01-insertion)
add_subdirectory(tasks/task-02-selection-bubble)
add_subdirectory(tasks/task-03-shell)
add_subdirectory(src)

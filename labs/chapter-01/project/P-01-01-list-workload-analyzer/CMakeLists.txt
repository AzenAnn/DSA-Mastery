cmake_minimum_required(VERSION 3.25)
project(list_workload_lab LANGUAGES CXX)
if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LAB_USE_SOLUTION "Build the reviewed reference implementation" OFF)
if(LAB_USE_SOLUTION)
  set(LAB_TARGET_VARIANT "solution")
else()
  set(LAB_TARGET_VARIANT "student")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/.lab-cache/bin/${LAB_TARGET_VARIANT}")
foreach(CONFIGURATION DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIGURATION} "${PROJECT_SOURCE_DIR}/.lab-cache/bin/${LAB_TARGET_VARIANT}")
endforeach()
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_subdirectory(tasks/task-01-sequential-list)
add_subdirectory(tasks/task-02-linked-list)
add_subdirectory(tasks/task-03-workload-runner)

cmake_minimum_required(VERSION 3.25)
project(dorm105 LANGUAGES CXX)
if(NOT DEFINED CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LAB_USE_SOLUTION "Build reference implementations" OFF)
set(LAB_IMPL student)
if(LAB_USE_SOLUTION)
  set(LAB_IMPL solution)
endif()
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/.lab-cache/bin/${LAB_IMPL}")
foreach(CONFIGURATION DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIGURATION}
      "${PROJECT_SOURCE_DIR}/.lab-cache/bin/${LAB_IMPL}")
endforeach()
enable_testing()
add_subdirectory(tasks/task-01-students)
add_subdirectory(tasks/task-02-delivery)
add_subdirectory(tasks/task-03-pending)
add_subdirectory(tasks/task-04-final)
